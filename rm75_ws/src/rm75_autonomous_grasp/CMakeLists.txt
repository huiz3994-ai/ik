cmake_minimum_required(VERSION 3.0.2)
project(rm75_autonomous_grasp)
add_compile_options(-std=c++14 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp rospy sensor_msgs geometry_msgs std_msgs std_srvs actionlib actionlib_msgs control_msgs trajectory_msgs moveit_core moveit_ros_planning moveit_ros_planning_interface moveit_msgs tf2_ros tf2_geometry_msgs pcl_ros pcl_conversions rm_msgs gazebo_ros gazebo_msgs message_generation)
find_package(gazebo REQUIRED)
add_action_files(FILES Grasp.action)
generate_messages(DEPENDENCIES actionlib_msgs std_msgs)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS message_runtime roscpp moveit_ros_planning_interface)
include_directories(include ${catkin_INCLUDE_DIRS} ${GAZEBO_INCLUDE_DIRS})
link_directories(${GAZEBO_LIBRARY_DIRS})
add_library(rm75_task grasp/grasp_planner.cpp planning/moveit_planner.cpp planning/trajectory_optimizer.cpp gripper/gripper.cpp)
add_dependencies(rm75_task ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
target_link_libraries(rm75_task ${catkin_LIBRARIES})
add_executable(grasp_node state_machine/grasp_state_machine.cpp)
add_dependencies(grasp_node ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
target_link_libraries(grasp_node rm75_task ${catkin_LIBRARIES})
add_executable(obstacle_map planning/obstacle_map.cpp)
target_link_libraries(obstacle_map ${catkin_LIBRARIES})
add_executable(trajectory_bridge execution/trajectory_bridge.cpp)
add_dependencies(trajectory_bridge ${catkin_EXPORTED_TARGETS})
target_link_libraries(trajectory_bridge ${catkin_LIBRARIES})
add_library(rm75_sim_gripper SHARED gripper/sim_gripper_plugin.cpp)
target_link_libraries(rm75_sim_gripper ${catkin_LIBRARIES} ${GAZEBO_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
 find_package(rostest REQUIRED)
 add_rostest(test/bridge.test DEPENDENCIES trajectory_bridge)
 catkin_add_gtest(test_collision test/test_collision.cpp)
 if(TARGET test_collision)
 target_link_libraries(test_collision rm75_task ${catkin_LIBRARIES})
 endif()
endif()
catkin_install_python(PROGRAMS scripts/scene_manager.py scripts/run_grasp.py scripts/integration_test.py scripts/preflight.py scripts/init_sim.py scripts/dynamic_test.py scripts/planner_benchmark.py test/test_bridge.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(TARGETS rm75_task grasp_node obstacle_map trajectory_bridge rm75_sim_gripper ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
install(DIRECTORY launch config urdf worlds docs test DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
