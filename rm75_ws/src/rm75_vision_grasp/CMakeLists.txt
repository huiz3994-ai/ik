cmake_minimum_required(VERSION 3.0.2)
project(rm75_vision_grasp)
add_compile_options(-std=c++14 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs geometry_msgs std_msgs image_transport cv_bridge message_filters tf2_ros tf2_geometry_msgs)
find_package(OpenCV REQUIRED)
catkin_package(INCLUDE_DIRS include LIBRARIES rm75_vision CATKIN_DEPENDS roscpp sensor_msgs geometry_msgs)
include_directories(include ${catkin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
add_library(rm75_vision vision/detector.cpp vision/depth_pose.cpp)
target_link_libraries(rm75_vision ${catkin_LIBRARIES} ${OpenCV_LIBRARIES})
add_executable(vision_node vision/vision_node.cpp)
target_link_libraries(vision_node rm75_vision ${catkin_LIBRARIES} ${OpenCV_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
 catkin_add_gtest(test_depth test/test_depth.cpp)
 if(TARGET test_depth)
 target_link_libraries(test_depth rm75_vision ${catkin_LIBRARIES})
 endif()
endif()
install(TARGETS rm75_vision vision_node ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
